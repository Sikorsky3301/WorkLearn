================================================================================
                            AUTH CONTEXT API
================================================================================


  loginDirect(email, password)
    → success: { success: true, role }
    → fail:    { error: 'Invalid email or password.' }

  loginUniversity(rollNo, password)
    → rejects mentors: { error: 'Mentors must log in at the Mentor Login page.' }
    → success: { success: true, role: 'university_student' }
    → fail:    { error: 'Invalid Roll Number or password.' }

  loginMentor(mentorId, password)
    → rejects students: { error: 'This portal is for mentors only...' }
    → success: { success: true, role: 'class_mentor' }
    → fail:    { error: 'Invalid Mentor ID or password.' }

  logout()           → clears user, redirect to respective login page
  hasFeature(name)   → checks role defaults then user.unlockedFeatures[]
  unlockFeature(name)→ appends to user.unlockedFeatures[]
